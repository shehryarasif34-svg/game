  <h2>Random Number Guessing Game (1 to 10)</h2>
  <p>Page open hotay hi game pop-up mein shuru ho jayegi. Dobara khelne ke liye page refresh (F5) karein.</p>

  <script>
    const secret = Math.floor(Math.random() * 10) + 1;
    let guess = Number(prompt("Guess a number between 1 and 10:"));

    while (guess !== secret) {
      if (guess < secret) {
        guess = Number(prompt("Too low! Try again:"));
      } else {
        guess = Number(prompt("Too high! Try again:"));
      }
    }

    alert("Mubarak ho! Aap ne sahi number guess kar liya: " + secret);
  </script>
